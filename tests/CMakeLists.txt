# ============================================================================
# sameboy_core â€” SameBoy Core compiled as a C static library for test use.
# Compiled with the SAME clang-cl toolchain Enginemon already uses (build_clang/).
# The main `build/` uses MSVC cl.exe. This target is added to BOTH build trees;
# only the build_clang tree will compile it correctly (clang-cl supports
# __attribute__((visibility)) and #include_next needed by SameBoy).
#
# No SameBoy source files are modified.
# Windows/ shims (part of SameBoy tree, unmodified) provide vasprintf and alloca.
# GB_VERSION requires escaped string literal: \"1.0.3\" in compile definitions.
# ============================================================================
# SameBoy Core: built when SAMEBOY_ORACLE=ON (clang-cl build tree only).
# Gate on an explicit option so the condition doesn't depend on compiler-detection
# timing under VCPKG chainload, where CMAKE_C_COMPILER may not be populated yet.
# ============================================================================
option(SAMEBOY_ORACLE "Build SameBoy Core oracle targets (clang-cl build tree only)" OFF)
if(ENGINEMON_ENABLE_ENGINE AND SAMEBOY_ORACLE)
    set(SAMEBOY_ROOT "${CMAKE_SOURCE_DIR}/references/SameBoy")

    # ─── SameBoy source-tree integrity check (configure-time) ─────────────────
    # Capture the current commit and git-status of references/SameBoy.
    # These are embedded into sameboy_version.h so the runtime can verify both.
    execute_process(
        COMMAND git rev-parse HEAD
        WORKING_DIRECTORY "${SAMEBOY_ROOT}"
        OUTPUT_VARIABLE  SAMEBOY_HEAD_COMMIT
        ERROR_QUIET OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(
        COMMAND git status --porcelain
        WORKING_DIRECTORY "${SAMEBOY_ROOT}"
        OUTPUT_VARIABLE  SAMEBOY_DIRTY_STATUS
        ERROR_QUIET OUTPUT_STRIP_TRAILING_WHITESPACE)

    # Write out sameboy_version.h with embedded commit and clean status.
    # This file is generated into the source tree so it is stable across builds.
    # It is already tracked in git -- we overwrite it only if values changed.
    set(_SB_VER_FILE "${CMAKE_SOURCE_DIR}/tests/crystal_differential/sameboy_version.h")
    file(WRITE "${_SB_VER_FILE}"
"#pragma once\n"
"/* Test-only SameBoy identity header -- regenerated by CMake at configure time.\n"
" * Do NOT edit by hand. Pin the SameBoy subtree commit before editing. */\n"
"\n"
"#ifndef GB_VERSION\n"
"#define GB_VERSION \"1.0.3\"\n"
"#endif\n"
"\n"
"#ifndef GB_COMMIT\n"
"/* Full SHA-1 of references/SameBoy HEAD at the last configure run. */\n"
"#define GB_COMMIT \"${SAMEBOY_HEAD_COMMIT}\"\n"
"#endif\n"
"\n"
"#ifndef GB_DIRTY\n"
"/* Non-empty string if references/SameBoy has uncommitted tracked changes.\n"
" * Empty string means the working tree is clean. */\n"
"#define GB_DIRTY \"${SAMEBOY_DIRTY_STATUS}\"\n"
"#endif\n"
"\n"
"#ifndef GB_COPYRIGHT_YEAR\n"
"#define GB_COPYRIGHT_YEAR \"2026\"\n"
"#endif\n")

    message(STATUS "SameBoy commit: ${SAMEBOY_HEAD_COMMIT}")
    if(SAMEBOY_DIRTY_STATUS)
        message(WARNING "SameBoy working tree has modifications:\n${SAMEBOY_DIRTY_STATUS}")
    endif()

    add_library(sameboy_core STATIC
        ${SAMEBOY_ROOT}/Core/apu.c
        ${SAMEBOY_ROOT}/Core/camera.c
        ${SAMEBOY_ROOT}/Core/display.c
        ${SAMEBOY_ROOT}/Core/gb.c
        ${SAMEBOY_ROOT}/Core/joypad.c
        ${SAMEBOY_ROOT}/Core/mbc.c
        ${SAMEBOY_ROOT}/Core/memory.c
        ${SAMEBOY_ROOT}/Core/printer.c
        ${SAMEBOY_ROOT}/Core/random.c
        ${SAMEBOY_ROOT}/Core/rumble.c
        ${SAMEBOY_ROOT}/Core/save_state.c
        ${SAMEBOY_ROOT}/Core/sgb.c
        ${SAMEBOY_ROOT}/Core/sm83_cpu.c
        ${SAMEBOY_ROOT}/Core/timing.c
        ${SAMEBOY_ROOT}/Core/workboy.c
        ${SAMEBOY_ROOT}/Windows/crt.c
        ${SAMEBOY_ROOT}/Windows/dirent.c
        ${SAMEBOY_ROOT}/Windows/math.c
        ${SAMEBOY_ROOT}/Windows/stdio.c
        ${SAMEBOY_ROOT}/Windows/utf8_compat.c
    )
    # All sources are C99/C11 â€” force C language on each TU
    set_source_files_properties(
        ${SAMEBOY_ROOT}/Core/apu.c
        ${SAMEBOY_ROOT}/Core/camera.c
        ${SAMEBOY_ROOT}/Core/display.c
        ${SAMEBOY_ROOT}/Core/gb.c
        ${SAMEBOY_ROOT}/Core/joypad.c
        ${SAMEBOY_ROOT}/Core/mbc.c
        ${SAMEBOY_ROOT}/Core/memory.c
        ${SAMEBOY_ROOT}/Core/printer.c
        ${SAMEBOY_ROOT}/Core/random.c
        ${SAMEBOY_ROOT}/Core/rumble.c
        ${SAMEBOY_ROOT}/Core/save_state.c
        ${SAMEBOY_ROOT}/Core/sgb.c
        ${SAMEBOY_ROOT}/Core/sm83_cpu.c
        ${SAMEBOY_ROOT}/Core/timing.c
        ${SAMEBOY_ROOT}/Core/workboy.c
        ${SAMEBOY_ROOT}/Windows/crt.c
        ${SAMEBOY_ROOT}/Windows/dirent.c
        ${SAMEBOY_ROOT}/Windows/math.c
        ${SAMEBOY_ROOT}/Windows/stdio.c
        ${SAMEBOY_ROOT}/Windows/utf8_compat.c
        PROPERTIES LANGUAGE CXX  # Force CXX compiler (clang-cl); /TC in compile options forces C parsing
    )
    target_include_directories(sameboy_core
        PUBLIC  ${SAMEBOY_ROOT}           # consumers: #include "Core/gb.h"
        PRIVATE ${SAMEBOY_ROOT}/Windows   # Windows/ shims shadow system headers
                ${CMAKE_SOURCE_DIR}/tests/crystal_differential  # sameboy_version.h
    )
    target_compile_definitions(sameboy_core PRIVATE
        GB_INTERNAL
        GB_DISABLE_TIMEKEEPING
        GB_DISABLE_REWIND
        GB_DISABLE_DEBUGGER
        GB_DISABLE_CHEATS
        GB_DISABLE_CHEAT_SEARCH
        _USE_MATH_DEFINES
        # GB_VERSION and GB_COPYRIGHT_YEAR are provided via sameboy_version.h
        # (passing string literals via -D on MSVC/clang-cl command lines requires
        # backslash-escaping that CMake does not consistently translate; a header is safer)
    )
    if(MSVC)
        # clang-cl: /TC = treat as C, /std:c11. Suppress warnings from SameBoy internals.
        # /FI: force-include sameboy_version.h which defines GB_VERSION and GB_COPYRIGHT_YEAR
        #      as proper C string literals (safer than -D with escaped quotes).
        # NOTE: sources are marked LANGUAGE CXX so clang-cl is used (not cl.exe).
        # /TC overrides /TP; -clang:-std=c11 overrides -clang:-std=c++23 from global CXX flags.
        target_compile_options(sameboy_core PRIVATE
            /TC /std:c11 -clang:-std=c11
            "/FI${CMAKE_SOURCE_DIR}/tests/crystal_differential/sameboy_version.h"
            /wd4244 /wd4267 /wd4200 /wd4146 /wd4018 /wd4996 /wd4133
            /wd4090 /wd4101 /wd4047 /wd4028 /wd4245 /wd4005 /wd4715 /wd4334
        )
        # -Wno-include-next-absolute-path is a clang flag; only add it for clang-cl.
        if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
            target_compile_options(sameboy_core PRIVATE
                -Wno-include-next-absolute-path
            )
        endif()
    endif()

    # ============================================================================
    # Oracle anti-confirmation canaries (crystal_battle_diff --fault).
    #
    # Each canary is a PAIR of CTest entries:
    #   *_baseline  — unmodified case; PASS if oracle returns MATCH
    #   *_mutated   — deliberate Enginemon fault injected; PASS if oracle returns
    #                 ENGINEMON_MISMATCH (i.e. the bug is correctly detected)
    #
    # The test suite PASSES when mutation detection works.
    # The test suite FAILS if a mutated case ever returns MATCH.
    #
    # Fault bitmask bits (--fault <hex>):
    #   0x01  Return      enemy_hp -= 1  (simulates +1 damage)
    #   0x02  Recover     player_hp -= 1 (simulates -1 heal)
    #   0x04  Haze        player_stages[ATK] += 1 (stage not reset)
    #   0x08  DragonRage  enemy_hp -= 1  (simulates +1 damage)
    #   0x10  Return      rng_bytes_consumed += 1 (phantom extra RNG byte)
    #
    # Label: oracle-diff (runs with the rest of the oracle suite).
    # ============================================================================
    if(ENGINEMON_ROM_PATH AND ENGINEMON_SYM_PATH)
        set(_CBD $<TARGET_FILE:crystal_battle_diff>)
        set(_ROM "${ENGINEMON_ROM_PATH}")
        set(_SYM "${ENGINEMON_SYM_PATH}")

        # -- Return damage +1 --
        add_test(NAME oracle_canary_return_damage_baseline
            COMMAND ${_CBD} "${_ROM}" "${_SYM}" --move 216
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
        set_tests_properties(oracle_canary_return_damage_baseline PROPERTIES
            LABELS "oracle-diff" TIMEOUT 30
            PASS_REGULAR_EXPRESSION "MATCH: *1")

        add_test(NAME oracle_canary_return_damage_mutated
            COMMAND ${_CBD} "${_ROM}" "${_SYM}" --move 216 --fault 0x01
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
        set_tests_properties(oracle_canary_return_damage_mutated PROPERTIES
            LABELS "oracle-diff" TIMEOUT 30
            PASS_REGULAR_EXPRESSION "enemy_hp Crystal=[0-9]+ Enginemon=[0-9]+")

        # -- Recover heal -1 --
        add_test(NAME oracle_canary_recover_heal_baseline
            COMMAND ${_CBD} "${_ROM}" "${_SYM}" --move 105
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
        set_tests_properties(oracle_canary_recover_heal_baseline PROPERTIES
            LABELS "oracle-diff" TIMEOUT 30
            PASS_REGULAR_EXPRESSION "MATCH: *1")

        add_test(NAME oracle_canary_recover_heal_mutated
            COMMAND ${_CBD} "${_ROM}" "${_SYM}" --move 105 --fault 0x02
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
        set_tests_properties(oracle_canary_recover_heal_mutated PROPERTIES
            LABELS "oracle-diff" TIMEOUT 30
            PASS_REGULAR_EXPRESSION "player_hp Crystal=[0-9]+ Enginemon=[0-9]+")

        # -- Haze ATK stage +1 --
        add_test(NAME oracle_canary_haze_stage_baseline
            COMMAND ${_CBD} "${_ROM}" "${_SYM}" --move 114
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
        set_tests_properties(oracle_canary_haze_stage_baseline PROPERTIES
            LABELS "oracle-diff" TIMEOUT 30
            PASS_REGULAR_EXPRESSION "MATCH: *1")

        add_test(NAME oracle_canary_haze_stage_mutated
            COMMAND ${_CBD} "${_ROM}" "${_SYM}" --move 114 --fault 0x04
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
        set_tests_properties(oracle_canary_haze_stage_mutated PROPERTIES
            LABELS "oracle-diff" TIMEOUT 30
            PASS_REGULAR_EXPRESSION "player_stage\\.ATK Crystal=0 Enginemon=1")

        # -- DragonRage damage +1 --
        add_test(NAME oracle_canary_dragonrage_damage_baseline
            COMMAND ${_CBD} "${_ROM}" "${_SYM}" --move 82
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
        set_tests_properties(oracle_canary_dragonrage_damage_baseline PROPERTIES

            PASS_REGULAR_EXPRESSION "MATCH: *1")

        add_test(NAME oracle_canary_dragonrage_damage_mutated

            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
        set_tests_properties(oracle_canary_dragonrage_damage_mutated PROPERTIES
            LABELS "oracle-diff" TIMEOUT 30
            PASS_REGULAR_EXPRESSION "enemy_hp Crystal=[0-9]+ Enginemon=[0-9]+")

        # -- Return extra RNG byte --
        add_test(NAME oracle_canary_return_rng_baseline
            COMMAND ${_CBD} "${_ROM}" "${_SYM}" --move 216
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
        set_tests_properties(oracle_canary_return_rng_baseline PROPERTIES
            LABELS "oracle-diff" TIMEOUT 30
            PASS_REGULAR_EXPRESSION "MATCH: *1")

        add_test(NAME oracle_canary_return_rng_mutated
            COMMAND ${_CBD} "${_ROM}" "${_SYM}" --move 216 --fault 0x10

    endif()

# =============================================================================
# CTest registration
# =============================================================================
# Standalone tests (no ROM required) are always registered when engine is enabled.
# ROM-dependent tests are registered only when ENGINEMON_ROM_PATH is set.
#
# Usage:
#   cmake -B build -DENGINEMON_ROM_PATH="path/to/crystal.gbc"
#   ctest --test-dir build -j8 --output-on-failure
#
# Labels:
#   engine-only  -- no Crystal compiler IR, no ROM needed; safe at full parallelism
#   compiler-ir  -- Crystal compiler/IR variants; high memory per TU
#   standalone   -- no ROM, no compiler IR
#   rom-required -- requires ENGINEMON_ROM_PATH
#   oracle       -- heavyweight (2.5s); run last in parallel batch
# =============================================================================
